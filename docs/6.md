## تنزيلات متعددة بالتوازي: الحل الـMVP الأنظف
على **نفس الاتصال** لا يمكن تنزيل ملفين “بالتوازي الحقيقي” لأن السيرفر سيرسل bytes بشكل متسلسل على نفس الـSocket.  
لذلك الـMVP الصحيح للتوازي هو:

- **اتصال دائم (Control Channel)**: للأوامر الخفيفة + الإشعارات (`LIST_USERS`, `LIST_MY_FILES`, `INCOMING_FILE`…)
- **اتصال مستقل لكل تنزيل (Download Channel)**: Socket جديد لكل ملف ⇒ تنزيلات متعددة بالتوازي بدون أي تداخل قراءة.

هذا يجعل الكود أبسط وأكثر استقرارًا.

## 1) تعديل `ClientConnection` لدعم Parallel Downloads عبر Socket مستقل لكل تنزيل

```java
package com.hamza.filetransfer.client;

import com.hamza.filetransfer.shared.Limits;
import com.hamza.filetransfer.shared.Protocol;
import javafx.application.Platform;

import java.io.*;
import java.net.Socket;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;
import java.util.function.BiConsumer;
import java.util.function.Consumer;

public class ClientConnection implements Closeable {

    private final String host;
    private final int port;
    private final String username;

    private final Socket socket;
    private final DataInputStream in;
    private final DataOutputStream out;
    private final Object writeLock = new Object();

    private volatile boolean running = true;

    // UI callbacks
    private Consumer<String> onError = msg -> {};
    private Consumer<List<String>> onUsers = users -> {};
    private Consumer<List<InboxFile>> onMyFiles = files -> {};
    private Consumer<InboxFile> onIncoming = file -> {};

    public ClientConnection(String host, int port, String username) throws Exception {
        this.host = host;
        this.port = port;
        this.username = username;

        this.socket = new Socket(host, port);
        this.in = new DataInputStream(new BufferedInputStream(socket.getInputStream()));
        this.out = new DataOutputStream(new BufferedOutputStream(socket.getOutputStream()));

        // HELLO
        synchronized (writeLock) {
            out.writeUTF(Protocol.HELLO);
            out.writeUTF(username);
            out.flush();
        }

        String resp = in.readUTF();
        if (Protocol.HELLO_TAKEN.equals(resp)) {
            throw new IllegalStateException("Username already online.");
        }
        if (!Protocol.HELLO_OK.equals(resp)) {
            throw new IllegalStateException("Unexpected HELLO response: " + resp);
        }

        Thread listener = new Thread(this::listenLoop, "client-control-listener");
        listener.setDaemon(true);
        listener.start();
    }

    public void setOnError(Consumer<String> onError) { this.onError = onError; }
    public void setOnUsers(Consumer<List<String>> onUsers) { this.onUsers = onUsers; }
    public void setOnMyFiles(Consumer<List<InboxFile>> onMyFiles) { this.onMyFiles = onMyFiles; }
    public void setOnIncoming(Consumer<InboxFile> onIncoming) { this.onIncoming = onIncoming; }

    private void listenLoop() {
        try {
            while (running) {
                String type = in.readUTF();

                if (Protocol.ERROR.equals(type)) {
                    String msg = in.readUTF();
                    Platform.runLater(() -> onError.accept(msg));

                } else if (Protocol.USERS.equals(type)) {
                    int n = in.readInt();
                    List<String> users = new ArrayList<>(n);
                    for (int i = 0; i < n; i++) users.add(in.readUTF());
                    Platform.runLater(() -> onUsers.accept(users));

                } else if (Protocol.MY_FILES.equals(type)) {
                    int n = in.readInt();
                    List<InboxFile> files = new ArrayList<>(n);
                    for (int i = 0; i < n; i++) {
                        String fileId = in.readUTF();
                        String fileName = in.readUTF();
                        long size = in.readLong();
                        String fromUser = in.readUTF();
                        String ts = in.readUTF();
                        files.add(new InboxFile(fileId, fileName, size, fromUser, ts));
                    }
                    Platform.runLater(() -> onMyFiles.accept(files));

                } else if (Protocol.INCOMING_FILE.equals(type)) {
                    String fileId = in.readUTF();
                    String fromUser = in.readUTF();
                    String fileName = in.readUTF();
                    long size = in.readLong();
                    String ts = in.readUTF();
                    InboxFile f = new InboxFile(fileId, fileName, size, fromUser, ts);
                    Platform.runLater(() -> onIncoming.accept(f));

                } else if (Protocol.SEND_OK.equals(type)) {
                    String fileId = in.readUTF();
                    // optional: notify UI

                } else if (Protocol.FILE.equals(type)) {
                    // لا نتوقع تنزيل على قناة التحكم
                    Platform.runLater(() -> onError.accept("Protocol error: FILE received on control channel"));

                } else {
                    Platform.runLater(() -> onError.accept("Unknown server message: " + type));
                }
            }
        } catch (EOFException ignored) {
            if (running) Platform.runLater(() -> onError.accept("Disconnected"));
        } catch (Exception e) {
            if (running) Platform.runLater(() -> onError.accept("Disconnected: " + e.getMessage()));
        }
    }

    public void requestUsers() throws Exception {
        synchronized (writeLock) {
            out.writeUTF(Protocol.LIST_USERS);
            out.flush();
        }
    }

    public void requestMyFiles() throws Exception {
        synchronized (writeLock) {
            out.writeUTF(Protocol.LIST_MY_FILES);
            out.flush();
        }
    }

    public void sendFile(String toUser, Path file, BiConsumer<Long, Long> onProgress) throws Exception {
        long size = Files.size(file);
        if (size > Limits.MAX_FILE_BYTES) {
            throw new IllegalArgumentException("File exceeds 200MB limit.");
        }

        String fileName = file.getFileName().toString();

        synchronized (writeLock) {
            out.writeUTF(Protocol.SEND_FILE);
            out.writeUTF(toUser);
            out.writeUTF(fileName);
            out.writeLong(size);
            out.flush();

            try (InputStream fileIn = Files.newInputStream(file)) {
                byte[] buffer = new byte[Limits.BUFFER_SIZE];
                long sent = 0;
                int read;
                while ((read = fileIn.read(buffer)) != -1) {
                    out.write(buffer, 0, read);
                    sent += read;

                    long sentFinal = sent;
                    Platform.runLater(() -> onProgress.accept(sentFinal, size));
                }
                out.flush();
            }
        }
    }

    /**
     * Parallel download (MVP):
     * - opens a NEW socket per download
     * - performs HELLO with same username
     * - sends DOWNLOAD fileId
     * - receives FILE + bytes and writes to disk
     */
    public void startParallelDownload(String fileId, Path targetDirOrFile,
                                      BiConsumer<Long, Long> onProgress,
                                      Consumer<String> onStatus) {

        Thread t = new Thread(() -> {
            Platform.runLater(() -> onStatus.accept("CONNECTING"));

            try (Socket s = new Socket(host, port);
                 DataInputStream din = new DataInputStream(new BufferedInputStream(s.getInputStream()));
                 DataOutputStream dout = new DataOutputStream(new BufferedOutputStream(s.getOutputStream()))) {

                // HELLO on download channel (same username)
                dout.writeUTF(Protocol.HELLO);
                dout.writeUTF(username);
                dout.flush();

                String helloResp = din.readUTF();
                if (!Protocol.HELLO_OK.equals(helloResp)) {
                    throw new IOException("Download channel HELLO failed: " + helloResp);
                }

                // DOWNLOAD
                dout.writeUTF(Protocol.DOWNLOAD);
                dout.writeUTF(fileId);
                dout.flush();

                String type = din.readUTF();
                if (Protocol.ERROR.equals(type)) {
                    String msg = din.readUTF();
                    throw new IOException(msg);
                }
                if (!Protocol.FILE.equals(type)) {
                    throw new IOException("Unexpected response: " + type);
                }

                String fileName = din.readUTF();
                long fileSize = din.readLong();

                Path finalTarget = Files.isDirectory(targetDirOrFile)
                        ? targetDirOrFile.resolve(fileName)
                        : targetDirOrFile;

                Files.createDirectories(finalTarget.getParent());

                Platform.runLater(() -> onStatus.accept("DOWNLOADING"));

                try (OutputStream fileOut = Files.newOutputStream(finalTarget,
                        StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING)) {

                    byte[] buffer = new byte[Limits.BUFFER_SIZE];
                    long received = 0;

                    while (received < fileSize) {
                        int toRead = (int) Math.min(buffer.length, fileSize - received);
                        int read = din.read(buffer, 0, toRead);
                        if (read == -1) throw new EOFException("Disconnected during download");

                        fileOut.write(buffer, 0, read);
                        received += read;

                        long doneFinal = received;
                        Platform.runLater(() -> onProgress.accept(doneFinal, fileSize));
                    }
                }

                Platform.runLater(() -> onStatus.accept("DONE"));

            } catch (Exception ex) {
                Platform.runLater(() -> onStatus.accept("FAILED: " + ex.getMessage()));
            }
        }, "download-" + fileId);

        t.setDaemon(true);
        t.start();
    }

    @Override
    public void close() throws IOException {
        running = false;
        socket.close();
    }
}
```

## 2) تعديل `ClientApp` لاستدعاء `startParallelDownload` بدل النسخة السابقة

```java
package com.hamza.filetransfer.client;

import javafx.application.Application;
import javafx.application.Platform;
import javafx.collections.*;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.control.cell.ProgressBarTableCell;
import javafx.scene.layout.*;
import javafx.stage.DirectoryChooser;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.nio.file.Path;
import java.util.List;

public class ClientApp extends Application {

    private ClientConnection conn;

    private final ObservableList<String> users = FXCollections.observableArrayList();
    private final ObservableList<InboxFile> inbox = FXCollections.observableArrayList();

    private final ObservableList<SimpleTransferRow> transfers = FXCollections.observableArrayList();

    @Override
    public void start(Stage stage) {
        TextField tfHost = new TextField("127.0.0.1");
        TextField tfPort = new TextField("5000");
        TextField tfUser = new TextField("user1");
        Button btnConnect = new Button("Connect");

        Label lblStatus = new Label("Not connected");

        HBox top = new HBox(10, new Label("Host:"), tfHost, new Label("Port:"), tfPort, new Label("User:"), tfUser, btnConnect, lblStatus);
        top.setPadding(new Insets(10));

        ListView<String> usersView = new ListView<>(users);
        usersView.setPrefWidth(200);

        TableView<InboxFile> inboxTable = new TableView<>(inbox);
        inboxTable.setColumnResizePolicy(TableView.CONSTRAINED_RESIZE_POLICY_ALL_COLUMNS);

        TableColumn<InboxFile, String> fName = new TableColumn<>("File");
        fName.setCellValueFactory(c -> c.getValue().fileNameProperty());

        TableColumn<InboxFile, String> fFrom = new TableColumn<>("From");
        fFrom.setCellValueFactory(c -> c.getValue().fromUserProperty());

        TableColumn<InboxFile, Number> fSize = new TableColumn<>("Size");
        fSize.setCellValueFactory(c -> c.getValue().sizeProperty());

        inboxTable.getColumns().addAll(fName, fFrom, fSize);

        Button btnRefreshUsers = new Button("Refresh Users");
        Button btnRefreshFiles = new Button("Refresh My Files");

        Button btnChoose = new Button("Choose File");
        Label lblChosen = new Label("No file selected");
        Button btnSend = new Button("Send to selected user");

        VBox left = new VBox(10, new Label("Online Users"), usersView, btnRefreshUsers);
        left.setPadding(new Insets(10));

        VBox center = new VBox(10, new Label("My Inbox"), inboxTable, new HBox(10, btnRefreshFiles));
        center.setPadding(new Insets(10));

        VBox bottomSend = new VBox(8, new HBox(10, btnChoose, lblChosen), btnSend);
        bottomSend.setPadding(new Insets(10));

        TableView<SimpleTransferRow> transfersTable = new TableView<>(transfers);
        transfersTable.setColumnResizePolicy(TableView.CONSTRAINED_RESIZE_POLICY_ALL_COLUMNS);

        TableColumn<SimpleTransferRow, String> tName = new TableColumn<>("Transfer");
        tName.setCellValueFactory(c -> c.getValue().nameProperty());

        TableColumn<SimpleTransferRow, Double> tProg = new TableColumn<>("Progress");
        tProg.setCellValueFactory(c -> c.getValue().progressProperty().asObject());
        tProg.setCellFactory(ProgressBarTableCell.forTableColumn());

        TableColumn<SimpleTransferRow, String> tStatus = new TableColumn<>("Status");
        tStatus.setCellValueFactory(c -> c.getValue().statusProperty());

        transfersTable.getColumns().addAll(tName, tProg, tStatus);

        BorderPane root = new BorderPane();
        root.setTop(top);
        root.setLeft(left);
        root.setCenter(center);

        VBox right = new VBox(10, new Label("Transfers"), transfersTable, new Separator(), new Label("Send"), bottomSend);
        right.setPadding(new Insets(10));
        right.setPrefWidth(380);
        root.setRight(right);

        inboxTable.setRowFactory(tv -> {
            TableRow<InboxFile> row = new TableRow<>();
            row.setOnMouseClicked(e -> {
                if (e.getClickCount() == 2 && !row.isEmpty()) {
                    InboxFile f = row.getItem();
                    chooseAndDownload(stage, f);
                }
            });
            return row;
        });

        final Path[] chosen = new Path[1];

        btnChoose.setOnAction(a -> {
            FileChooser fc = new FileChooser();
            var file = fc.showOpenDialog(stage);
            if (file != null) {
                chosen[0] = file.toPath();
                lblChosen.setText(chosen[0].toString());
            }
        });

        btnConnect.setOnAction(a -> {
            try {
                conn = new ClientConnection(tfHost.getText().trim(), Integer.parseInt(tfPort.getText().trim()), tfUser.getText().trim());
                lblStatus.setText("Connected");

                conn.setOnError(msg -> lblStatus.setText("Error: " + msg));
                conn.setOnUsers(this::setUsers);
                conn.setOnMyFiles(this::setInbox);
                conn.setOnIncoming(f -> {
                    inbox.add(f);
                    lblStatus.setText("Incoming: " + f.fileNameProperty().get());
                });

                conn.requestUsers();
                conn.requestMyFiles();

            } catch (Exception ex) {
                lblStatus.setText("Connect failed: " + ex.getMessage());
            }
        });

        btnRefreshUsers.setOnAction(a -> {
            if (conn == null) return;
            try { conn.requestUsers(); } catch (Exception ex) { lblStatus.setText(ex.getMessage()); }
        });

        btnRefreshFiles.setOnAction(a -> {
            if (conn == null) return;
            try { conn.requestMyFiles(); } catch (Exception ex) { lblStatus.setText(ex.getMessage()); }
        });

        btnSend.setOnAction(a -> {
            if (conn == null) return;
            String to = usersView.getSelectionModel().getSelectedItem();
            if (to == null) {
                lblStatus.setText("Select a user first.");
                return;
            }
            if (chosen[0] == null) {
                lblStatus.setText("Choose a file first.");
                return;
            }

            SimpleTransferRow row = new SimpleTransferRow("SEND " + chosen[0].getFileName() + " -> " + to);
            row.setStatus("SENDING");
            transfers.add(row);

            Thread t = new Thread(() -> {
                try {
                    conn.sendFile(to, chosen[0], row::setProgress);
                    Platform.runLater(() -> row.setStatus("DONE"));
                } catch (Exception ex) {
                    Platform.runLater(() -> row.setStatus("FAILED: " + ex.getMessage()));
                }
            }, "send-task");
            t.setDaemon(true);
            t.start();
        });

        stage.setTitle("File Transfer - Client");
        stage.setScene(new Scene(root, 1200, 600));
        stage.show();
    }

    private void setUsers(List<String> list) {
        users.setAll(list);
    }

    private void setInbox(List<InboxFile> list) {
        inbox.setAll(list);
    }

    private void chooseAndDownload(Stage stage, InboxFile f) {
        if (conn == null) return;

        DirectoryChooser dc = new DirectoryChooser();
        dc.setTitle("Choose download folder");
        var dir = dc.showDialog(stage);
        if (dir == null) return;

        SimpleTransferRow row = new SimpleTransferRow("DOWNLOAD " + f.fileNameProperty().get());
        row.setStatus("QUEUED");
        transfers.add(row);

        conn.startParallelDownload(
                f.fileIdProperty().get(),
                dir.toPath(),
                row::setProgress,
                row::setStatus
        );
    }
}
```

## 3) ملاحظة مهمة جدًا للسيرفر (username مكرر على قناة تنزيل)
حاليًا السيرفر عنده شرط: إذا `username` Online بالفعل → يرجع `HELLO_TAKEN`.  
وبما أننا سنفتح Socket جديد للتنزيل بنفس username، لازم السيرفر يسمح بأكثر من اتصال لنفس المستخدم.

### أبسط MVP تعديل على السيرفر
بدل `username -> اتصال واحد`، نخلي:
- **قناة التحكم فقط** هي التي تُسجل في `online` (للإشعارات)
- أما اتصالات التنزيل/التحميل المؤقتة لا تُسجل في `online`

لكن لأن سيرفرك الحالي يسجل كل اتصال بعد HELLO، لازم نعدّله قليلًا:
- نجعل `HELLO` يرجع OK دائمًا
- ونسجل في `online` فقط إذا كانت الجلسة “control” (مثلاً أول أمر بعد HELLO يكون `CONTROL`)

إذا تحب، أعطيك تعديل MVP سريع جدًا:
- العميل يرسل بعد HELLO مباشرة: `MODE CONTROL` أو `MODE TRANSFER`
- السيرفر يسجل online فقط لـ CONTROL

قلّي: تفضّل هذا التعديل (MODE)؟  
إذا نعم، سأرسل لك تعديلات جاهزة على `FileTransferServer` و`Protocol` بحيث تشتغل تنزيلات متعددة فورًا بدون تعارض.
