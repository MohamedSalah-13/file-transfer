## شكل أقرب لتطبيقات الشات + زر Light/Dark (AtlantaFX)

سنحوّل واجهة **Client** من “جداول” إلى تخطيط شبيه بالشات:
- **يسار:** قائمة المستخدمين + Search
- **وسط:** “محادثة/Inbox” كـ **فقاعات (bubbles)** (ListView مع Cell مخصص)
- **أسفل:** شريط إرسال (Attach + Send)
- **أعلى:** Topbar فيه **Toggle Light/Dark**

> لن نغير الشبكة الآن؛ فقط UI. (يعني “المحادثة” هنا هي عرض ملفات واردة/مرسلة كرسائل ملف).

## 1) أضف ThemeManager (تبديل Light/Dark)

**Create file:** `src/main/java/com/hamza/filetransfer/client/ThemeManager.java`
```java
package com.hamza.filetransfer.client;

import atlantafx.base.theme.PrimerDark;
import atlantafx.base.theme.PrimerLight;
import javafx.application.Application;

public final class ThemeManager {
    private static boolean dark = false;

    private ThemeManager() {}

    public static void applyLight() {
        dark = false;
        Application.setUserAgentStylesheet(new PrimerLight().getUserAgentStylesheet());
    }

    public static void applyDark() {
        dark = true;
        Application.setUserAgentStylesheet(new PrimerDark().getUserAgentStylesheet());
    }

    public static boolean isDark() {
        return dark;
    }

    public static void toggle() {
        if (dark) applyLight();
        else applyDark();
    }
}
```

## 3) تعديل `ClientApp` إلى تخطيط شات + Toggle Light/Dark

> التغيير الأكبر: استبدال `TableView inboxTable` بـ `ListView<InboxFile>` مع “bubble cell”.

```java
package com.hamza.filetransfer.client;

import javafx.application.Application;
import javafx.application.Platform;
import javafx.collections.*;
import javafx.collections.transformation.FilteredList;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.*;
import javafx.stage.DirectoryChooser;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.nio.file.Path;
import java.util.List;

public class ClientApp extends Application {

    private ClientConnection conn;

    private final ObservableList<String> users = FXCollections.observableArrayList();
    private final ObservableList<InboxFile> inbox = FXCollections.observableArrayList();
    private final ObservableList<SimpleTransferRow> transfers = FXCollections.observableArrayList();

    @Override
    public void start(Stage stage) {
        // Default theme
        ThemeManager.applyLight();

        TextField tfHost = new TextField("127.0.0.1");
        TextField tfPort = new TextField("5000");
        TextField tfUser = new TextField("user1");
        Button btnConnect = new Button("Connect");

        Label lblStatus = new Label("Not connected");

        ToggleButton themeToggle = new ToggleButton("Dark");
        themeToggle.setOnAction(e -> {
            ThemeManager.toggle();
            themeToggle.setSelected(ThemeManager.isDark());
            themeToggle.setText(ThemeManager.isDark() ? "Light" : "Dark");
        });

        HBox top = new HBox(
                10,
                new Label("Host:"), tfHost,
                new Label("Port:"), tfPort,
                new Label("User:"), tfUser,
                btnConnect,
                new Region(),
                themeToggle,
                lblStatus
        );
        HBox.setHgrow(top.getChildren().get(7), Priority.ALWAYS); // Region spacer
        top.setAlignment(Pos.CENTER_LEFT);
        top.getStyleClass().add("topbar");

        // ===== Left sidebar (Users + search) =====
        TextField tfSearchUsers = new TextField();
        tfSearchUsers.setPromptText("Search users...");

        FilteredList<String> filteredUsers = new FilteredList<>(users, u -> true);
        tfSearchUsers.textProperty().addListener((obs, old, q) -> {
            String query = q == null ? "" : q.trim().toLowerCase();
            filteredUsers.setPredicate(u -> query.isEmpty() || u.toLowerCase().contains(query));
        });

        ListView<String> usersView = new ListView<>(filteredUsers);
        usersView.setPrefWidth(260);
        usersView.getSelectionModel().setSelectionMode(SelectionMode.SINGLE);

        Button btnRefreshUsers = new Button("Refresh");

        VBox left = new VBox(10,
                new Label("Online"),
                tfSearchUsers,
                usersView,
                btnRefreshUsers
        );
        left.getStyleClass().add("sidebar");

        // ===== Center chat surface (Inbox as bubbles) =====
        Label chatTitle = new Label("Inbox");
        chatTitle.getStyleClass().add("chat-title");

        Label chatSub = new Label("Double-click a file bubble to download");
        chatSub.getStyleClass().add("chat-subtitle");

        ListView<InboxFile> chatList = new ListView<>(inbox);
        chatList.setCellFactory(lv -> new InboxBubbleCell());

        // double click download
        chatList.setOnMouseClicked(e -> {
            if (e.getClickCount() == 2) {
                InboxFile f = chatList.getSelectionModel().getSelectedItem();
                if (f != null) chooseAndDownload(stage, f);
            }
        });

        Button btnRefreshFiles = new Button("Refresh");
        HBox chatHeader = new HBox(10, chatTitle, new Region(), btnRefreshFiles);
        HBox.setHgrow(chatHeader.getChildren().get(1), Priority.ALWAYS);

        VBox center = new VBox(10, chatHeader, chatSub, chatList);
        center.getStyleClass().add("chat-surface");

        // ===== Right: Transfers + Composer (Send) =====
        TableView<SimpleTransferRow> transfersTable = new TableView<>(transfers);
        transfersTable.setColumnResizePolicy(TableView.CONSTRAINED_RESIZE_POLICY_ALL_COLUMNS);

        TableColumn<SimpleTransferRow, String> tName = new TableColumn<>("Transfer");
        tName.setCellValueFactory(c -> c.getValue().nameProperty());

        TableColumn<SimpleTransferRow, Double> tProg = new TableColumn<>("Progress");
        tProg.setCellValueFactory(c -> c.getValue().progressProperty().asObject());
        tProg.setCellFactory(tc -> new ProgressBarTableCell<>());

        TableColumn<SimpleTransferRow, String> tStatus = new TableColumn<>("Status");
        tStatus.setCellValueFactory(c -> c.getValue().statusProperty());

        transfersTable.getColumns().addAll(tName, tProg, tStatus);

        Button btnChoose = new Button("Attach");
        Label lblChosen = new Label("No file");
        Button btnSend = new Button("Send");

        final Path[] chosen = new Path[1];

        btnChoose.setOnAction(a -> {
            FileChooser fc = new FileChooser();
            var file = fc.showOpenDialog(stage);
            if (file != null) {
                chosen[0] = file.toPath();
                lblChosen.setText(chosen[0].getFileName().toString());
            }
        });

        btnSend.setOnAction(a -> {
            if (conn == null) return;

            String to = usersView.getSelectionModel().getSelectedItem();
            if (to == null) {
                lblStatus.setText("Select a user first.");
                return;
            }
            if (chosen[0] == null) {
                lblStatus.setText("Attach a file first.");
                return;
            }

            SimpleTransferRow row = new SimpleTransferRow("SEND " + chosen[0].getFileName() + " -> " + to);
            row.setStatus("SENDING");
            transfers.add(row);

            Thread t = new Thread(() -> {
                try {
                    conn.sendFile(to, chosen[0], row::setProgress);
                    Platform.runLater(() -> row.setStatus("DONE"));
                } catch (Exception ex) {
                    Platform.runLater(() -> row.setStatus("FAILED: " + ex.getMessage()));
                }
            }, "send-task");
            t.setDaemon(true);
            t.start();
        });

        HBox composer = new HBox(10, btnChoose, lblChosen, new Region(), btnSend);
        HBox.setHgrow(composer.getChildren().get(3), Priority.ALWAYS);
        composer.setAlignment(Pos.CENTER_LEFT);
        composer.getStyleClass().add("composer");

        VBox right = new VBox(10,
                new Label("Transfers"),
                transfersTable,
                new Separator(),
                new Label("Send"),
                composer
        );
        right.getStyleClass().add("card");
        right.setPrefWidth(420);

        // ===== Root layout =====
        BorderPane root = new BorderPane();
        root.setTop(top);

        BorderPane content = new BorderPane();
        content.setLeft(left);
        content.setCenter(center);
        content.setRight(right);
        BorderPane.setMargin(content, new Insets(12));
        root.setCenter(content);

        // Connect logic
        btnConnect.setOnAction(a -> {
            try {
                conn = new ClientConnection(tfHost.getText().trim(), Integer.parseInt(tfPort.getText().trim()), tfUser.getText().trim());
                lblStatus.setText("Connected");

                conn.setOnError(msg -> lblStatus.setText("Error: " + msg));
                conn.setOnUsers(this::setUsers);
                conn.setOnMyFiles(this::setInbox);
                conn.setOnIncoming(f -> {
                    inbox.add(0, f);
                    lblStatus.setText("Incoming: " + f.fileNameProperty().get());
                });

                conn.requestUsers();
                conn.requestMyFiles();

            } catch (Exception ex) {
                lblStatus.setText("Connect failed: " + ex.getMessage());
            }
        });

        btnRefreshUsers.setOnAction(a -> {
            if (conn == null) return;
            try { conn.requestUsers(); } catch (Exception ex) { lblStatus.setText(ex.getMessage()); }
        });

        btnRefreshFiles.setOnAction(a -> {
            if (conn == null) return;
            try { conn.requestMyFiles(); } catch (Exception ex) { lblStatus.setText(ex.getMessage()); }
        });

        Scene scene = new Scene(root, 1280, 720);
        scene.getStylesheets().add(getClass().getResource("/styles/app.css").toExternalForm());

        stage.setTitle("File Transfer Chat - Client");
        stage.setScene(scene);
        stage.show();
    }

    private void setUsers(List<String> list) {
        users.setAll(list);
    }

    private void setInbox(List<InboxFile> list) {
        inbox.setAll(list);
    }

    private void chooseAndDownload(Stage stage, InboxFile f) {
        if (conn == null) return;

        DirectoryChooser dc = new DirectoryChooser();
        dc.setTitle("Choose download folder");
        var dir = dc.showDialog(stage);
        if (dir == null) return;

        SimpleTransferRow row = new SimpleTransferRow("DOWNLOAD " + f.fileNameProperty().get());
        row.setStatus("QUEUED");
        transfers.add(row);

        conn.startParallelDownload(
                f.fileIdProperty().get(),
                dir.toPath(),
                row::setProgress,
                row::setStatus
        );
    }

    /**
     * Bubble-like cell (chat style).
     */
    private static final class InboxBubbleCell extends ListCell<InboxFile> {
        @Override
        protected void updateItem(InboxFile item, boolean empty) {
            super.updateItem(item, empty);

            if (empty || item == null) {
                setGraphic(null);
                setText(null);
                return;
            }

            Label title = new Label(item.fileNameProperty().get());
            title.getStyleClass().add("bubble-title");

            Label meta = new Label("From: " + item.fromUserProperty().get() + "  •  " + humanSize(item.sizeProperty().get()));
            meta.getStyleClass().add("bubble-meta");

            VBox box = new VBox(4, title, meta);

            // inbound bubble (you can add outbound later when you store "sent")
            StackPane bubble = new StackPane(box);
            bubble.getStyleClass().addAll("bubble", "bubble-in");
            bubble.setMaxWidth(Double.MAX_VALUE);

            // align like chat (incoming left)
            HBox row = new HBox(bubble);
            row.setAlignment(Pos.CENTER_LEFT);

            setGraphic(row);
        }

        private static String humanSize(long bytes) {
            double b = bytes;
            String[] units = {"B", "KB", "MB", "GB"};
            int i = 0;
            while (b >= 1024 && i < units.length - 1) {
                b /= 1024;
                i++;
            }
            return String.format("%.1f %s", b, units[i]);
        }
    }
}
```
