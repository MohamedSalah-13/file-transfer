## سنقسم المشروع إلى تطبيقين JavaFX داخل نفس مشروع Maven
- **Server App**: واجهة + سيرفر TCP + عرض التحويلات مع Progress.
- **Client App**: واجهة + اتصال دائم بالسيرفر + إرسال/استقبال + Progress.

سأجهز لك **هيكل شغال (Functional MVP)** يدعم:
- Offline inbox (يُحفظ على السيرفر داخل مجلد المستخدم)
- إرسال مباشر لمستخدم محدد عبر السيرفر
- `LIST_USERS`, `LIST_MY_FILES`, `DOWNLOAD`
- حد ملف **200MB** (رفض قبل الاستقبال)

    private final StringProperty fileName = new SimpleStringProperty();
    private final LongProperty totalBytes = new SimpleLongProperty();
    private final LongProperty doneBytes = new SimpleLongProperty();

        this.toUser.set(to);

    public StringProperty fileNameProperty() { return fileName; }
    public LongProperty totalBytesProperty() { return totalBytes; }
    public LongProperty doneBytesProperty() { return doneBytes; }
    public DoubleProperty progressProperty() { return progress; }
    public StringProperty statusProperty() { return status; }

### 4.2 السيرفر نفسه (يدعم Offline + حد 200MB)
**Create file:** `src/main/java/com/hamza/filetransfer/server/FileTransferServer.java`
```java
package com.hamza.filetransfer.server;

import com.hamza.filetransfer.shared.Limits;
import com.hamza.filetransfer.shared.Protocol;

import java.io.*;
import java.net.ServerSocket;
import java.net.Socket;
import java.nio.file.*;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Consumer;

public class FileTransferServer {

    public record OnlineClient(String username, DataOutputStream out, Object lock) {}

    private final int port;
    private final Path usersDir;
    private final Map<String, OnlineClient> online = new ConcurrentHashMap<>();
    private final Consumer<TransferSession> onNewSession;

    public FileTransferServer(int port, Path usersDir, Consumer<TransferSession> onNewSession) {
        this.port = port;
        this.usersDir = usersDir;
        this.onNewSession = onNewSession;
    }

    public void startAsync() {
        Thread t = new Thread(this::run, "server-acceptor");
        t.setDaemon(true);
        t.start();
    }

    private void run() {
        try {
            Files.createDirectories(usersDir);
            try (ServerSocket serverSocket = new ServerSocket(port)) {
                while (true) {
                    Socket s = serverSocket.accept();
                    new Thread(() -> handleClient(s), "client-" + s.getRemoteSocketAddress()).start();
                }
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    private void handleClient(Socket socket) {
        String username = null;

        try (socket;
             DataInputStream in = new DataInputStream(new BufferedInputStream(socket.getInputStream()));
             DataOutputStream out = new DataOutputStream(new BufferedOutputStream(socket.getOutputStream()))) {

            Object lock = new Object();

            // أول رسالة لازم تكون HELLO
            String cmd = in.readUTF();
            if (!Protocol.HELLO.equals(cmd)) {
                sendError(out, lock, "Expected HELLO");
                return;
            }
            username = in.readUTF().trim();
            if (username.isBlank()) {
                sendError(out, lock, "Empty username");
                return;
            }

            if (online.putIfAbsent(username, new OnlineClient(username, out, lock)) != null) {
                synchronized (lock) {
                    out.writeUTF(Protocol.HELLO_TAKEN);
                    out.flush();
                }
                return;
            }

            Files.createDirectories(userInboxDir(username));

            synchronized (lock) {
                out.writeUTF(Protocol.HELLO_OK);
                out.flush();
            }

            // Loop للأوامر
            while (true) {
                String c = in.readUTF();

                if (Protocol.LIST_USERS.equals(c)) {
                    handleListUsers(out, lock);

                } else if (Protocol.SEND_FILE.equals(c)) {
                    handleSendFile(username, in, out, lock);

                } else if (Protocol.LIST_MY_FILES.equals(c)) {
                    handleListMyFiles(username, out, lock);

                } else if (Protocol.DOWNLOAD.equals(c)) {
                    handleDownload(username, in, out, lock);

                } else {
                    sendError(out, lock, "Unknown command: " + c);
                }
            }

        } catch (EOFException ignored) {
            // client disconnected
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            if (username != null) {
                online.remove(username);
            }
        }
    }

    private void handleListUsers(DataOutputStream out, Object lock) throws IOException {
        List<String> users = new ArrayList<>(online.keySet());
        Collections.sort(users);

        synchronized (lock) {
            out.writeUTF(Protocol.USERS);
            out.writeInt(users.size());
            for (String u : users) out.writeUTF(u);
            out.flush();
        }
    }

    private void handleSendFile(String fromUser, DataInputStream in, DataOutputStream out, Object lock) throws Exception {
        String toUser = in.readUTF().trim();
        String fileName = in.readUTF();
        long fileSize = in.readLong();

        if (fileSize < 0 || fileSize > Limits.MAX_FILE_BYTES) {
            sendError(out, lock, "File too large. Max is 200MB.");
            // مهم: لمنع desync، الأفضل إنهاء الاتصال بعد رفض حجم كبير
            throw new IOException("Rejected oversized file");
        }

        String fileId = UUID.randomUUID().toString();
        Instant ts = Instant.now();

        TransferSession session = new TransferSession(fromUser, toUser, fileName, fileSize);
        session.setStatus("RECEIVING");
        onNewSession.accept(session);

        Path inbox = userInboxDir(toUser);
        Files.createDirectories(inbox);

        Path bin = inbox.resolve(fileId + ".bin");
        Path meta = inbox.resolve(fileId + ".meta");

        // استلام bytes إلى bin
        try (OutputStream fileOut = Files.newOutputStream(bin, StandardOpenOption.CREATE_NEW)) {
            byte[] buffer = new byte[Limits.BUFFER_SIZE];
            long received = 0;

            while (received < fileSize) {
                int toRead = (int) Math.min(buffer.length, fileSize - received);
                int read = in.read(buffer, 0, toRead);
                if (read == -1) throw new EOFException("Disconnected during upload");
                fileOut.write(buffer, 0, read);
                received += read;

                session.setDoneBytes(received);
            }
        }

        session.setStatus("STORED");

        // اكتب meta (Properties بسيطة)
        String metaContent =
                "fileId=" + fileId + "\n" +
                "fromUser=" + fromUser + "\n" +
                "toUser=" + toUser + "\n" +
                "fileName=" + fileName + "\n" +
                "fileSize=" + fileSize + "\n" +
                "timestamp=" + ts + "\n";
        Files.writeString(meta, metaContent, StandardOpenOption.CREATE_NEW);

        // رد للمرسل
        synchronized (lock) {
            out.writeUTF(Protocol.SEND_OK);
            out.writeUTF(fileId);
            out.flush();
        }

        // إشعار فوري لو المستقبل Online
        OnlineClient target = online.get(toUser);
        if (target != null) {
            synchronized (target.lock()) {
                target.out().writeUTF(Protocol.INCOMING_FILE);
                target.out().writeUTF(fileId);
                target.out().writeUTF(fromUser);
                target.out().writeUTF(fileName);
                target.out().writeLong(fileSize);
                target.out().writeUTF(ts.toString());
                target.out().flush();
            }
        }

        session.setStatus("DONE");
    }

    private void handleListMyFiles(String username, DataOutputStream out, Object lock) throws Exception {
        Path inbox = userInboxDir(username);
        Files.createDirectories(inbox);

        List<Path> metas = new ArrayList<>();
        try (DirectoryStream<Path> ds = Files.newDirectoryStream(inbox, "*.meta")) {
            for (Path p : ds) metas.add(p);
        }

        synchronized (lock) {
            out.writeUTF(Protocol.MY_FILES);
            out.writeInt(metas.size());
            for (Path meta : metas) {
                Properties pr = new Properties();
                try (Reader r = Files.newBufferedReader(meta)) {
                    pr.load(r);
                }
                out.writeUTF(pr.getProperty("fileId"));
                out.writeUTF(pr.getProperty("fileName"));
                out.writeLong(Long.parseLong(pr.getProperty("fileSize")));
                out.writeUTF(pr.getProperty("fromUser"));
                out.writeUTF(pr.getProperty("timestamp"));
            }
            out.flush();
        }
    }

    private void handleDownload(String username, DataInputStream in, DataOutputStream out, Object lock) throws Exception {
        String fileId = in.readUTF();

        Path inbox = userInboxDir(username);
        Path meta = inbox.resolve(fileId + ".meta");
        Path bin = inbox.resolve(fileId + ".bin");

        if (!Files.exists(meta) || !Files.exists(bin)) {
            sendError(out, lock, "Not found");
            return;
        }

        Properties pr = new Properties();
        try (Reader r = Files.newBufferedReader(meta)) {
            pr.load(r);
        }

        String fileName = pr.getProperty("fileName");
        long fileSize = Long.parseLong(pr.getProperty("fileSize"));

        synchronized (lock) {
            out.writeUTF(Protocol.FILE);
            out.writeUTF(fileName);
            out.writeLong(fileSize);
            out.flush();
        }

        // إرسال bytes
        try (InputStream fileIn = Files.newInputStream(bin)) {
            byte[] buffer = new byte[Limits.BUFFER_SIZE];
            long sent = 0;
            int read;
            while ((read = fileIn.read(buffer)) != -1) {
                out.write(buffer, 0, read);
                sent += read;
            }
            out.flush();
        }
    }

    private void sendError(DataOutputStream out, Object lock, String message) throws IOException {
        synchronized (lock) {
            out.writeUTF(Protocol.ERROR);
            out.writeUTF(message);
            out.flush();
        }
    }

    private Path userInboxDir(String username) {
        return usersDir.resolve(username).resolve("inbox");
    }
}
```

        TableColumn<TransferSession, String> toCol = new TableColumn<>("To");

        statusCol.setCellValueFactory(c -> c.getValue().statusProperty());

        table.getColumns().addAll(fromCol, toCol, fileCol, progCol, statusCol);

        root.setTop(info);

        stage.setScene(new Scene(root, 900, 500));

### 5.2 عميل الشبكة (اتصال دائم + listener للإشعارات)
**Create file:** `src/main/java/com/hamza/filetransfer/client/ClientConnection.java`
```java
package com.hamza.filetransfer.client;

import com.hamza.filetransfer.shared.Limits;
import com.hamza.filetransfer.shared.Protocol;
import javafx.application.Platform;

import java.io.*;
import java.net.Socket;
import java.nio.file.*;
import java.util.ArrayList;
import java.util.List;
import java.util.function.BiConsumer;
import java.util.function.Consumer;

public class ClientConnection implements Closeable {

    private final Socket socket;
    private final DataInputStream in;
    private final DataOutputStream out;
    private final Object writeLock = new Object();

    private volatile boolean running = true;

    // UI callbacks
    private Consumer<String> onError = msg -> {};
    private Consumer<List<String>> onUsers = users -> {};
    private Consumer<List<InboxFile>> onMyFiles = files -> {};
    private Consumer<InboxFile> onIncoming = file -> {};

    public ClientConnection(String host, int port, String username) throws Exception {
        this.socket = new Socket(host, port);
        this.in = new DataInputStream(new BufferedInputStream(socket.getInputStream()));
        this.out = new DataOutputStream(new BufferedOutputStream(socket.getOutputStream()));

        // HELLO
        synchronized (writeLock) {
            out.writeUTF(Protocol.HELLO);
            out.writeUTF(username);
            out.flush();
        }

        String resp = in.readUTF();
        if (Protocol.HELLO_TAKEN.equals(resp)) {
            throw new IllegalStateException("Username already online.");
        }
        if (!Protocol.HELLO_OK.equals(resp)) {
            throw new IllegalStateException("Unexpected HELLO response: " + resp);
        }

        Thread listener = new Thread(this::listenLoop, "client-listener");
        listener.setDaemon(true);
        listener.start();
    }

    public void setOnError(Consumer<String> onError) { this.onError = onError; }
    public void setOnUsers(Consumer<List<String>> onUsers) { this.onUsers = onUsers; }
    public void setOnMyFiles(Consumer<List<InboxFile>> onMyFiles) { this.onMyFiles = onMyFiles; }
    public void setOnIncoming(Consumer<InboxFile> onIncoming) { this.onIncoming = onIncoming; }

    private void listenLoop() {
        try {
            while (running) {
                String type = in.readUTF();

                if (Protocol.ERROR.equals(type)) {
                    String msg = in.readUTF();
                    Platform.runLater(() -> onError.accept(msg));

                } else if (Protocol.USERS.equals(type)) {
                    int n = in.readInt();
                    List<String> users = new ArrayList<>(n);
                    for (int i = 0; i < n; i++) users.add(in.readUTF());
                    Platform.runLater(() -> onUsers.accept(users));

                } else if (Protocol.MY_FILES.equals(type)) {
                    int n = in.readInt();
                    List<InboxFile> files = new ArrayList<>(n);
                    for (int i = 0; i < n; i++) {
                        String fileId = in.readUTF();
                        String fileName = in.readUTF();
                        long size = in.readLong();
                        String fromUser = in.readUTF();
                        String ts = in.readUTF();
                        files.add(new InboxFile(fileId, fileName, size, fromUser, ts));
                    }
                    Platform.runLater(() -> onMyFiles.accept(files));

                } else if (Protocol.INCOMING_FILE.equals(type)) {
                    String fileId = in.readUTF();
                    String fromUser = in.readUTF();
                    String fileName = in.readUTF();
                    long size = in.readLong();
                    String ts = in.readUTF();
                    InboxFile f = new InboxFile(fileId, fileName, size, fromUser, ts);
                    Platform.runLater(() -> onIncoming.accept(f));

                } else if (Protocol.SEND_OK.equals(type)) {
                    String fileId = in.readUTF();
                    // يمكن عرض Toast/Label لو تحب

                } else if (Protocol.FILE.equals(type)) {
                    // هذا يُقرأ داخل download(...) مباشرة (نمنع تداخل القراءات)
                    // لذلك هنا نعتبره بروتوكولياً خطأ إذا وصلنا
                    Platform.runLater(() -> onError.accept("Protocol error: FILE received in listener loop"));

                } else {
                    Platform.runLater(() -> onError.accept("Unknown server message: " + type));
                }
            }
        } catch (Exception e) {
            if (running) Platform.runLater(() -> onError.accept("Disconnected: " + e.getMessage()));
        }
    }

    public void requestUsers() throws Exception {
        synchronized (writeLock) {
            out.writeUTF(Protocol.LIST_USERS);
            out.flush();
        }
    }

    public void requestMyFiles() throws Exception {
        synchronized (writeLock) {
            out.writeUTF(Protocol.LIST_MY_FILES);
            out.flush();
        }
    }

    public void sendFile(String toUser, Path file, BiConsumer<Long, Long> onProgress) throws Exception {
        long size = Files.size(file);
        if (size > Limits.MAX_FILE_BYTES) {
            throw new IllegalArgumentException("File exceeds 200MB limit.");
        }

        String fileName = file.getFileName().toString();

        synchronized (writeLock) {
            out.writeUTF(Protocol.SEND_FILE);
            out.writeUTF(toUser);
            out.writeUTF(fileName);
            out.writeLong(size);
            out.flush();

            try (InputStream fileIn = Files.newInputStream(file)) {
                byte[] buffer = new byte[Limits.BUFFER_SIZE];
                long sent = 0;
                int read;
                while ((read = fileIn.read(buffer)) != -1) {
                    out.write(buffer, 0, read);
                    sent += read;
                    onProgress.accept(sent, size);
                }
                out.flush();
            }
        }
    }

    public void downloadTo(String fileId, Path targetFile, BiConsumer<Long, Long> onProgress) throws Exception {
        // ملاحظة مهمة: لأن listener thread يقرأ من نفس الـInputStream،
        // لازم نوقفه أو نمنع تداخله. لأجل MVP: سنوقف listener مؤقتاً أثناء download.
        // الأفضل لاحقاً: قناة رسائل + queue أو “مُنسّق قراءات” واحد.
        synchronized (writeLock) {
            out.writeUTF(Protocol.DOWNLOAD);
            out.writeUTF(fileId);
            out.flush();

            String type = in.readUTF();
            if (Protocol.ERROR.equals(type)) {
                String msg = in.readUTF();
                throw new IOException(msg);
            }
            if (!Protocol.FILE.equals(type)) {
                throw new IOException("Unexpected response: " + type);
            }

            String fileName = in.readUTF();
            long fileSize = in.readLong();

            Path finalTarget = Files.isDirectory(targetFile) ? targetFile.resolve(fileName) : targetFile;
            Files.createDirectories(finalTarget.getParent());

            try (OutputStream fileOut = Files.newOutputStream(finalTarget, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING)) {
                byte[] buffer = new byte[Limits.BUFFER_SIZE];
                long received = 0;
                while (received < fileSize) {
                    int toRead = (int) Math.min(buffer.length, fileSize - received);
                    int read = in.read(buffer, 0, toRead);
                    if (read == -1) throw new EOFException("Disconnected during download");
                    fileOut.write(buffer, 0, read);
                    received += read;
                    onProgress.accept(received, fileSize);
                }
            }
        }
    }

    @Override
    public void close() throws IOException {
        running = false;
        socket.close();
    }
}
```


> ملاحظة تقنية صغيرة: في الـMVP أعلاه، `downloadTo()` يقرأ من نفس `in` المستخدم في listener؛ لتجنب تضارب القراءات، جعلته داخل `synchronized(writeLock)` مع قراءة مباشرة. لاحقًا نرتّبها بشكل “Message Dispatcher” نظيف (سأساعدك فيها عندما تريد تحسين الجودة).

import javafx.application.Application;
import javafx.application.Platform;

import javafx.geometry.Insets;

        tName.setCellValueFactory(c -> c.getValue().nameProperty());

                Platform.runLater(() -> row.setStatus("FAILED: " + ex.getMessage()));
            }
        }, "download-task");
        t.setDaemon(true);
        t.start();
    }

package com.hamza.filetransfer.client;

import javafx.beans.property.*;

public class SimpleTransferRow {
    private final StringProperty name = new SimpleStringProperty();
    private final DoubleProperty progress = new SimpleDoubleProperty(0);
    private final StringProperty status = new SimpleStringProperty("STARTING");

    public SimpleTransferRow(String name) {
        this.name.set(name);
    }

    public StringProperty nameProperty() { return name; }
    public DoubleProperty progressProperty() { return progress; }
    public StringProperty statusProperty() { return status; }

    public void setProgress(long done, long total) {
        progress.set(total <= 0 ? 0 : (double) done / total);
    }

    public void setStatus(String s) { status.set(s); }
}

## ملاحظة مهمة لتحسين “الاستقرار” لاحقًا (بعد ما تتأكد أنه يعمل)
حاليًا `downloadTo()` يقرأ مباشرة من الـ`DataInputStream` بينما listener thread أيضًا يقرأ؛ وضعت حماية جزئية بـ `writeLock`، لكن الحل الأنظف هو:
- Thread واحد فقط يقرأ دائمًا (Dispatcher)
- يوزّع الرسائل على “انتظار” Futures للطلبات (DOWNLOAD/LIST…) + إشعارات (INCOMING)

إذا تحب، بعد ما تشغله وتأكد أن الـMVP يعمل عندك، سأحوّله لك إلى Dispatcher نظيف 100% بدون أي احتمال تداخل.
