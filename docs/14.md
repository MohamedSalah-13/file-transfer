1) **ترتيب قائمة المستخدمين تلقائيًا**:

package com.hamza.filetransfer.client;

import javafx.beans.property.*;

public class UserItem {
    private final StringProperty username = new SimpleStringProperty();
    private final StringProperty lastPreview = new SimpleStringProperty("");
    private final StringProperty lastTime = new SimpleStringProperty("");   // NEW
    private final LongProperty lastActivityMillis = new SimpleLongProperty(0); // NEW
    private final IntegerProperty unreadCount = new SimpleIntegerProperty(0);

    public UserItem(String username) {
        this.username.set(username);
    }

    public StringProperty usernameProperty() { return username; }
    public StringProperty lastPreviewProperty() { return lastPreview; }
    public StringProperty lastTimeProperty() { return lastTime; } // NEW
    public LongProperty lastActivityMillisProperty() { return lastActivityMillis; } // NEW
    public IntegerProperty unreadCountProperty() { return unreadCount; }

    public String getUsername() { return username.get(); }

    public void bumpUnread() { unreadCount.set(unreadCount.get() + 1); }
    public void clearUnread() { unreadCount.set(0); }

    public void setLastPreview(String preview) {
        lastPreview.set(preview == null ? "" : preview);
    }

    public void updateLastActivity(long millis, String timeLabel) {
        lastActivityMillis.set(Math.max(lastActivityMillis.get(), millis));
        if (timeLabel != null) lastTime.set(timeLabel);
    }
}

# 2) إضافة مساعد لتحديد نوع الملف + Preview prefix

**Create file:** `src/main/java/com/hamza/filetransfer/client/FileTypeUtil.java`
```java
package com.hamza.filetransfer.client;

import java.time.Instant;
import java.time.ZoneId;
import java.time.format.DateTimeFormatter;
import java.util.Locale;

public final class FileTypeUtil {

    private static final DateTimeFormatter TIME_FMT =
            DateTimeFormatter.ofPattern("HH:mm").withZone(ZoneId.systemDefault());

    private FileTypeUtil() {}

    public enum Kind { IMAGE, ARCHIVE, OTHER }

    public static Kind kindOf(String fileName) {
        String ext = extension(fileName);
        return switch (ext) {
            case "png", "jpg", "jpeg", "gif", "bmp", "webp" -> Kind.IMAGE;
            case "rar", "zip", "7z" -> Kind.ARCHIVE;
            default -> Kind.OTHER;
        };
    }

    public static String previewPrefixForFile(String fileName) {
        return switch (kindOf(fileName)) {
            case IMAGE -> "[IMG] ";
            case ARCHIVE -> "[BOOK] ";
            case OTHER -> "[FILE] ";
        };
    }

    public static String extension(String name) {
        if (name == null) return "";
        int i = name.lastIndexOf('.');
        if (i < 0 || i == name.length() - 1) return "";
        return name.substring(i + 1).toLowerCase(Locale.ROOT);
    }

    public static long parseIsoMillis(String isoInstant) {
        try {
            return Instant.parse(isoInstant).toEpochMilli();
        } catch (Exception e) {
            return 0L;
        }
    }

    public static String formatTime(long millis) {
        if (millis <= 0) return "";
        return TIME_FMT.format(Instant.ofEpochMilli(millis));
    }
}
```

    private void applyUsers(List<String> usernames) {
        for (String u : usernames) {
            userIndex.computeIfAbsent(u, key -> {
                UserItem item = new UserItem(key);
                userItems.add(item);
                return item;
            });
        }

        userItems.removeIf(item -> !usernames.contains(item.getUsername()));
        userIndex.keySet().retainAll(usernames);

        // trigger sort update
        userItems.setAll(userItems);
    }

    private void bumpUnreadAndPreview(String fromUser, String preview, long activityMillis) {
        UserItem item = userIndex.get(fromUser);
        if (item == null) return;

        item.setLastPreview(preview);

        String timeLabel = FileTypeUtil.formatTime(activityMillis);
        item.updateLastActivity(activityMillis, timeLabel);

        if (activePeer == null || !activePeer.equals(fromUser)) {
            item.bumpUnread();
        }
    }

    private void clearUnread(String user) {
        UserItem item = userIndex.get(user);
        if (item != null) item.clearUnread();
    }

    private void updateLastPreviewFromThread(String peer, List<ChatMessage> msgs) {
        if (peer == null) return;
        UserItem item = userIndex.get(peer);
        if (item == null || msgs == null || msgs.isEmpty()) return;

        ChatMessage last = msgs.get(msgs.size() - 1);

        String preview;
        if (last.typeProperty().get() == ChatMessage.Type.TEXT) {
            String t = last.textProperty().get();
            preview = (t == null) ? "" : t.trim();
            if (preview.length() > 40) preview = preview.substring(0, 40) + "...";
            preview = "[MSG] " + preview;
        } else {
            String name = last.fileNameProperty().get();
            preview = FileTypeUtil.previewPrefixForFile(name) + name;
        }

        long ms = FileTypeUtil.parseIsoMillis(last.timestampProperty().get());
        item.setLastPreview(preview);
        item.updateLastActivity(ms, FileTypeUtil.formatTime(ms));
    }

            preview.textProperty().bind(item.lastPreviewProperty());
            preview.getStyleClass().add("user-preview");

            badge.getStyleClass().add("badge");
            badge.textProperty().bind(item.unreadCountProperty().asString());
            badge.visibleProperty().bind(item.unreadCountProperty().greaterThan(0));
            badge.managedProperty().bind(badge.visibleProperty());

            HBox row = new HBox(10, textBox, badge);
            row.setAlignment(Pos.CENTER_LEFT);
            row.getStyleClass().add("user-row");
            HBox.setHgrow(textBox, Priority.ALWAYS);

            setGraphic(row);
        }
    }

        });
        b.getStyleClass().add("file-badge");
        return b;
    }
