## 2.1 تعديل `FileTransferServer.java`
- إضافة فرع `SEND_TEXT`
- `handleListThread` يرجّع:
    - `messageType` (`TEXT` أو `FILE`)
    - لو TEXT: يرسل `text`
    - لو FILE: يرسل `fileName`, `fileSize`, `downloadable`

```java
package com.hamza.filetransfer.server;

import com.hamza.filetransfer.shared.Limits;
import com.hamza.filetransfer.shared.Protocol;

import java.io.*;
import java.net.ServerSocket;
import java.net.Socket;
import java.nio.file.*;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Consumer;

public class FileTransferServer {

    // ... existing code ...

    private void handleClient(Socket socket) {
        String username = null;
        boolean isControl = false;

        try (socket;
             DataInputStream in = new DataInputStream(new BufferedInputStream(socket.getInputStream()));
             DataOutputStream out = new DataOutputStream(new BufferedOutputStream(socket.getOutputStream()))) {

            Object lock = new Object();

            // ... existing code ... HELLO + MODE ...

            while (true) {
                String c = in.readUTF();

                if (Protocol.LIST_USERS.equals(c)) {
                    if (!isControl) {
                        sendError(out, lock, "LIST_USERS allowed only on CONTROL connection");
                        continue;
                    }
                    handleListUsers(out, lock);

                } else if (Protocol.SEND_FILE.equals(c)) {
                    handleSendFile(username, in, out, lock);

                } else if (Protocol.SEND_TEXT.equals(c)) {
                    handleSendText(username, in, out, lock);

                } else if (Protocol.LIST_MY_FILES.equals(c)) {
                    if (!isControl) {
                        sendError(out, lock, "LIST_MY_FILES allowed only on CONTROL connection");
                        continue;
                    }
                    handleListMyFiles(username, out, lock);

                } else if (Protocol.LIST_THREAD.equals(c)) {
                    if (!isControl) {
                        sendError(out, lock, "LIST_THREAD allowed only on CONTROL connection");
                        continue;
                    }
                    handleListThread(username, in, out, lock);

                } else if (Protocol.DOWNLOAD.equals(c)) {
                    handleDownload(username, in, out, lock);

                } else {
                    sendError(out, lock, "Unknown command: " + c);
                }
            }

        } catch (EOFException ignored) {
            // client disconnected
        } catch (Exception e) {
            e.printStackTrace();
        } finally {
            if (username != null && isControl) {
                online.remove(username);
            }
        }
    }

    private void handleSendText(String fromUser, DataInputStream in, DataOutputStream out, Object lock) throws Exception {
        String toUser = in.readUTF().trim();
        String text = in.readUTF();

        if (text == null) text = "";
        text = text.trim();
        if (text.isEmpty()) {
            sendError(out, lock, "Empty text");
            return;
        }
        if (text.length() > 2000) {
            sendError(out, lock, "Text too long (max 2000 chars)");
            return;
        }

        String messageId = UUID.randomUUID().toString();
        Instant ts = Instant.now();
        long epochMillis = ts.toEpochMilli();

        // Thread meta for recipient (IN)
        writeThreadMetaText(toUser, fromUser, messageId, "IN", fromUser, toUser, text, ts.toString(), epochMillis);

        // Thread meta for sender (OUT)
        writeThreadMetaText(fromUser, toUser, messageId, "OUT", fromUser, toUser, text, ts.toString(), epochMillis);

        synchronized (lock) {
            out.writeUTF(Protocol.SEND_OK);
            out.writeUTF(messageId);
            out.flush();
        }

        // (اختياري) إشعار فوري للمستلم حتى يعمل refresh
        OnlineClient targetControl = online.get(toUser);
        if (targetControl != null) {
            synchronized (targetControl.lock()) {
                targetControl.out().writeUTF(Protocol.INCOMING_FILE);
                targetControl.out().writeUTF(messageId);
                targetControl.out().writeUTF(fromUser);
                targetControl.out().writeUTF("[text]");
                targetControl.out().writeLong(0L);
                targetControl.out().writeUTF(ts.toString());
                targetControl.out().flush();
            }
        }
    }

    private void handleListThread(String username, DataInputStream in, DataOutputStream out, Object lock) throws Exception {
        String peer = in.readUTF().trim();
        Path dir = userThreadDir(username, peer);
        Files.createDirectories(dir);

        List<Properties> messages = new ArrayList<>();
        try (DirectoryStream<Path> ds = Files.newDirectoryStream(dir, "*.meta")) {
            for (Path p : ds) {
                Properties pr = new Properties();
                try (Reader r = Files.newBufferedReader(p)) {
                    pr.load(r);
                }
                messages.add(pr);
            }
        }

        messages.sort(Comparator.comparingLong(p -> Long.parseLong(p.getProperty("timestampMillis", "0"))));

        synchronized (lock) {
            out.writeUTF(Protocol.THREAD_MESSAGES);
            out.writeUTF(peer);
            out.writeInt(messages.size());

            for (Properties pr : messages) {
                String messageType = pr.getProperty("messageType", "FILE"); // backward compatible

                out.writeUTF(pr.getProperty("messageId"));
                out.writeUTF(pr.getProperty("direction"));     // IN / OUT
                out.writeUTF(messageType);                     // TEXT / FILE
                out.writeUTF(pr.getProperty("fromUser"));
                out.writeUTF(pr.getProperty("toUser"));
                out.writeUTF(pr.getProperty("timestamp"));

                if ("TEXT".equals(messageType)) {
                    out.writeUTF(pr.getProperty("text", ""));
                } else {
                    out.writeUTF(pr.getProperty("fileName"));
                    out.writeLong(Long.parseLong(pr.getProperty("fileSize")));
                    out.writeBoolean(Boolean.parseBoolean(pr.getProperty("downloadable", "false")));
                }
            }
            out.flush();
        }
    }

    private void writeThreadMetaText(String ownerUser, String peerUser, String messageId,
                                     String direction, String fromUser, String toUser,
                                     String text, String timestamp, long timestampMillis) throws IOException {

        Path dir = userThreadDir(ownerUser, peerUser);
        Files.createDirectories(dir);

        Path meta = dir.resolve(messageId + ".meta");
        String content =
                "messageId=" + messageId + "\n" +
                "messageType=TEXT\n" +
                "direction=" + direction + "\n" +
                "fromUser=" + fromUser + "\n" +
                "toUser=" + toUser + "\n" +
                "timestamp=" + timestamp + "\n" +
                "timestampMillis=" + timestampMillis + "\n" +
                "text=" + escape(text) + "\n";

        Files.writeString(meta, content, StandardOpenOption.CREATE_NEW);
    }

    // Update existing writeThreadMeta(...) for FILE to include messageType
    private void writeThreadMeta(String ownerUser, String peerUser, String messageId,
                                 String direction, String fromUser, String toUser,
                                 String fileName, long fileSize,
                                 String timestamp, long timestampMillis,
                                 boolean downloadable) throws IOException {

        Path dir = userThreadDir(ownerUser, peerUser);
        Files.createDirectories(dir);

        Path meta = dir.resolve(messageId + ".meta");
        String content =
                "messageId=" + messageId + "\n" +
                "messageType=FILE\n" +
                "direction=" + direction + "\n" +
                "fromUser=" + fromUser + "\n" +
                "toUser=" + toUser + "\n" +
                "fileName=" + fileName + "\n" +
                "fileSize=" + fileSize + "\n" +
                "timestamp=" + timestamp + "\n" +
                "timestampMillis=" + timestampMillis + "\n" +
                "downloadable=" + downloadable + "\n";

        Files.writeString(meta, content, StandardOpenOption.CREATE_NEW);
    }

    private static String escape(String s) {
        // Properties format: we keep it simple and avoid newlines breaking the file
        return s.replace("\\", "\\\\").replace("\n", "\\n").replace("\r", "\\r");
    }

    // ... existing code ...
}
```

# 3) العميل: تحديث `ChatMessage` ليحمل TEXT أو FILE

```java
package com.hamza.filetransfer.client;

import javafx.beans.property.*;

public class ChatMessage {

    public enum Type { TEXT, FILE }

    private final StringProperty messageId = new SimpleStringProperty();
    private final ObjectProperty<Type> type = new SimpleObjectProperty<>(Type.FILE);

    private final BooleanProperty outgoing = new SimpleBooleanProperty(false);
    private final StringProperty fromUser = new SimpleStringProperty();
    private final StringProperty toUser = new SimpleStringProperty();
    private final StringProperty timestamp = new SimpleStringProperty();

    // TEXT
    private final StringProperty text = new SimpleStringProperty("");

    // FILE
    private final StringProperty fileName = new SimpleStringProperty("");
    private final LongProperty fileSize = new SimpleLongProperty(0);
    private final BooleanProperty downloadable = new SimpleBooleanProperty(false);

    private ChatMessage() {}

    public static ChatMessage text(String messageId, boolean outgoing, String fromUser, String toUser,
                                   String timestamp, String text) {
        ChatMessage m = new ChatMessage();
        m.messageId.set(messageId);
        m.type.set(Type.TEXT);
        m.outgoing.set(outgoing);
        m.fromUser.set(fromUser);
        m.toUser.set(toUser);
        m.timestamp.set(timestamp);
        m.text.set(text);
        return m;
    }

    public static ChatMessage file(String messageId, boolean outgoing, String fromUser, String toUser,
                                   String timestamp, String fileName, long fileSize, boolean downloadable) {
        ChatMessage m = new ChatMessage();
        m.messageId.set(messageId);
        m.type.set(Type.FILE);
        m.outgoing.set(outgoing);
        m.fromUser.set(fromUser);
        m.toUser.set(toUser);
        m.timestamp.set(timestamp);
        m.fileName.set(fileName);
        m.fileSize.set(fileSize);
        m.downloadable.set(downloadable);
        return m;
    }

    public StringProperty messageIdProperty() { return messageId; }
    public ObjectProperty<Type> typeProperty() { return type; }
    public BooleanProperty outgoingProperty() { return outgoing; }
    public StringProperty fromUserProperty() { return fromUser; }
    public StringProperty toUserProperty() { return toUser; }
    public StringProperty timestampProperty() { return timestamp; }

    public StringProperty textProperty() { return text; }

    public StringProperty fileNameProperty() { return fileName; }
    public LongProperty fileSizeProperty() { return fileSize; }
    public BooleanProperty downloadableProperty() { return downloadable; }
}
```

## 6.1 تعديل Bubble cell في `ClientApp`
اجعلها:
- إذا `TEXT`: يعرض `text`
- إذا `FILE`: يعرض fileName + meta + downloadable hint

```java
// ... existing code ...

    private static final class ChatBubbleCell extends ListCell<ChatMessage> {
        @Override
        protected void updateItem(ChatMessage item, boolean empty) {
            super.updateItem(item, empty);

            if (empty || item == null) {
                setGraphic(null);
                setText(null);
                return;
            }

            boolean out = item.outgoingProperty().get();

            Label title = new Label(out ? "You" : item.fromUserProperty().get());
            title.getStyleClass().add("bubble-title");

            Label body;
            Label meta;

            if (item.typeProperty().get() == ChatMessage.Type.TEXT) {
                body = new Label(item.textProperty().get());
                body.setWrapText(true);

                meta = new Label(item.timestampProperty().get());
                meta.getStyleClass().add("bubble-meta");
            } else {
                body = new Label(item.fileNameProperty().get());
                body.setWrapText(true);

                String metaText = humanSize(item.fileSizeProperty().get())
                        + "  •  " + item.timestampProperty().get()
                        + (item.downloadableProperty().get() ? "  •  double-click to download" : "");
                meta = new Label(metaText);
                meta.getStyleClass().add("bubble-meta");
            }

            VBox box = new VBox(4, title, body, meta);

            StackPane bubble = new StackPane(box);
            bubble.getStyleClass().add("bubble");
            bubble.getStyleClass().add(out ? "bubble-out" : "bubble-in");

            HBox row = new HBox(bubble);
            row.setAlignment(out ? Pos.CENTER_RIGHT : Pos.CENTER_LEFT);

            setGraphic(row);
        }

        private static String humanSize(long bytes) {
            double b = bytes;
            String[] units = {"B", "KB", "MB", "GB"};
            int i = 0;
            while (b >= 1024 && i < units.length - 1) {
                b /= 1024;
                i++;
            }
            return String.format("%.1f %s", b, units[i]);
        }
    }
```
