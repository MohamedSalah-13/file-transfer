## (B) Thumbnail حقيقي للصورة بعد التنزيل — “مثل واتساب” تمامًا

الفكرة:
- عند تنزيل ملف صورة بنجاح، نخزّن **مسار الملف المحلي** داخل الرسالة (`ChatMessage`) كـ `localPath`.
- الـBubble إذا كانت الرسالة **FILE + IMAGE + localPath موجود** ⇒ تعرض **ImageView thumbnail** داخل الفقاعة.
- هذا كله **Client-side فقط** (لا نغير السيرفر).

# 1) تحديث `ChatMessage` لإضافة `localPath`

```java
package com.hamza.filetransfer.client;

import javafx.beans.property.*;

import java.nio.file.Path;

public class ChatMessage {

    public enum Type { TEXT, FILE }

    private final StringProperty messageId = new SimpleStringProperty();
    private final ObjectProperty<Type> type = new SimpleObjectProperty<>(Type.FILE);

    private final BooleanProperty outgoing = new SimpleBooleanProperty(false);
    private final StringProperty fromUser = new SimpleStringProperty();
    private final StringProperty toUser = new SimpleStringProperty();
    private final StringProperty timestamp = new SimpleStringProperty();

    // TEXT
    private final StringProperty text = new SimpleStringProperty("");

    // FILE
    private final StringProperty fileName = new SimpleStringProperty("");
    private final LongProperty fileSize = new SimpleLongProperty(0);
    private final BooleanProperty downloadable = new SimpleBooleanProperty(false);

    // NEW: local downloaded file path (for thumbnails)
    private final ObjectProperty<Path> localPath = new SimpleObjectProperty<>(null);

    private ChatMessage() {}

    public static ChatMessage text(String messageId, boolean outgoing, String fromUser, String toUser,
                                   String timestamp, String text) {
        ChatMessage m = new ChatMessage();
        m.messageId.set(messageId);
        m.type.set(Type.TEXT);
        m.outgoing.set(outgoing);
        m.fromUser.set(fromUser);
        m.toUser.set(toUser);
        m.timestamp.set(timestamp);
        m.text.set(text);
        return m;
    }

    public static ChatMessage file(String messageId, boolean outgoing, String fromUser, String toUser,
                                   String timestamp, String fileName, long fileSize, boolean downloadable) {
        ChatMessage m = new ChatMessage();
        m.messageId.set(messageId);
        m.type.set(Type.FILE);
        m.outgoing.set(outgoing);
        m.fromUser.set(fromUser);
        m.toUser.set(toUser);
        m.timestamp.set(timestamp);
        m.fileName.set(fileName);
        m.fileSize.set(fileSize);
        m.downloadable.set(downloadable);
        return m;
    }

    public StringProperty messageIdProperty() { return messageId; }
    public ObjectProperty<Type> typeProperty() { return type; }
    public BooleanProperty outgoingProperty() { return outgoing; }
    public StringProperty fromUserProperty() { return fromUser; }
    public StringProperty toUserProperty() { return toUser; }
    public StringProperty timestampProperty() { return timestamp; }

    public StringProperty textProperty() { return text; }

    public StringProperty fileNameProperty() { return fileName; }
    public LongProperty fileSizeProperty() { return fileSize; }
    public BooleanProperty downloadableProperty() { return downloadable; }

    public ObjectProperty<Path> localPathProperty() { return localPath; }
}
```

# 2) تحديث `ClientConnection.startParallelDownload(...)` لإرجاع مسار الملف بعد الحفظ

نضيف parameter جديد: `Consumer<Path> onSaved`.

```java
package com.hamza.filetransfer.client;

// ... existing imports ...
import java.util.function.BiConsumer;
import java.util.function.Consumer;

public class ClientConnection implements Closeable {

    // ... existing code ...

    public void startParallelDownload(String fileId, Path targetDirOrFile,
                                      BiConsumer<Long, Long> onProgress,
                                      Consumer<String> onStatus,
                                      Consumer<Path> onSaved) {

        Thread t = new Thread(() -> {
            Platform.runLater(() -> onStatus.accept("CONNECTING"));

            try (Socket s = new Socket(host, port);
                 DataInputStream din = new DataInputStream(new BufferedInputStream(s.getInputStream()));
                 DataOutputStream dout = new DataOutputStream(new BufferedOutputStream(s.getOutputStream()))) {

                dout.writeUTF(Protocol.HELLO);
                dout.writeUTF(username);
                dout.flush();

                String helloResp = din.readUTF();
                if (!Protocol.HELLO_OK.equals(helloResp)) {
                    throw new IOException("Download channel HELLO failed: " + helloResp);
                }

                dout.writeUTF(Protocol.MODE);
                dout.writeUTF(Protocol.MODE_TRANSFER);
                dout.flush();

                dout.writeUTF(Protocol.DOWNLOAD);
                dout.writeUTF(fileId);
                dout.flush();

                String type = din.readUTF();
                if (Protocol.ERROR.equals(type)) {
                    String msg = din.readUTF();
                    throw new IOException(msg);
                }
                if (!Protocol.FILE.equals(type)) {
                    throw new IOException("Unexpected response: " + type);
                }

                String fileName = din.readUTF();
                long fileSize = din.readLong();

                Path finalTarget = Files.isDirectory(targetDirOrFile)
                        ? targetDirOrFile.resolve(fileName)
                        : targetDirOrFile;

                Files.createDirectories(finalTarget.getParent());

                Platform.runLater(() -> onStatus.accept("DOWNLOADING"));

                try (OutputStream fileOut = Files.newOutputStream(finalTarget,
                        StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING)) {

                    byte[] buffer = new byte[Limits.BUFFER_SIZE];
                    long received = 0;

                    while (received < fileSize) {
                        int toRead = (int) Math.min(buffer.length, fileSize - received);
                        int read = din.read(buffer, 0, toRead);
                        if (read == -1) throw new EOFException("Disconnected during download");

                        fileOut.write(buffer, 0, read);
                        received += read;

                        long doneFinal = received;
                        Platform.runLater(() -> onProgress.accept(doneFinal, fileSize));
                    }
                }

                Platform.runLater(() -> {
                    onSaved.accept(finalTarget);
                    onStatus.accept("DONE");
                });

            } catch (Exception ex) {
                Platform.runLater(() -> onStatus.accept("FAILED: " + ex.getMessage()));
            }
        }, "download-" + fileId);

        t.setDaemon(true);
        t.start();
    }

    // ... existing code ...
}
```


> لا تنسَ تعديل أي استدعاء قديم لـ `startParallelDownload(...)` ليضيف `onSaved`.

# 3) تحديث `ClientApp`: عند اكتمال التنزيل نربط الصورة بالرسالة ونحدث الـListView

سنعدل `chooseAndDownload(...)` بحيث:
- بعد `onSaved(path)`:
    - نبحث عن الرسالة في `chatMessages` بنفس `messageId`
    - نضع `localPath`
    - نعمل refresh للـListView (أو نعمل touch للـObservableList)

```java
package com.hamza.filetransfer.client;

import javafx.application.Application;
import javafx.application.Platform;
import javafx.collections.*;
import javafx.collections.transformation.FilteredList;
import javafx.collections.transformation.SortedList;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.Node;
import javafx.scene.control.*;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.layout.*;
import javafx.stage.DirectoryChooser;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.nio.file.Path;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class ClientApp extends Application {

    // ... existing code ...

    private ListView<ChatMessage> chatListRef; // NEW: for refresh after thumbnail

    @Override
    public void start(Stage stage) {
        // ... existing code ...

        ListView<ChatMessage> chatList = new ListView<>(chatMessages);
        this.chatListRef = chatList;
        chatList.setCellFactory(lv -> new ChatBubbleCell());

        // ... existing code ...
    }

    private void chooseAndDownload(Stage stage, String messageId, String displayName) {
        if (conn == null) return;

        DirectoryChooser dc = new DirectoryChooser();
        dc.setTitle("Choose download folder");
        var dir = dc.showDialog(stage);
        if (dir == null) return;

        SimpleTransferRow row = new SimpleTransferRow("DOWNLOAD " + displayName);
        row.setStatus("QUEUED");
        transfers.add(row);

        conn.startParallelDownload(
                messageId,
                dir.toPath(),
                row::setProgress,
                row::setStatus,
                savedPath -> {
                    // attach local path to the message (enables thumbnail)
                    ChatMessage msg = findMessageById(messageId);
                    if (msg != null) {
                        msg.localPathProperty().set(savedPath);
                        // refresh list cell rendering
                        if (chatListRef != null) chatListRef.refresh();
                    }
                }
        );
    }

    private ChatMessage findMessageById(String messageId) {
        for (ChatMessage m : chatMessages) {
            if (m.messageIdProperty().get().equals(messageId)) return m;
        }
        return null;
    }

    private static final class ChatBubbleCell extends ListCell<ChatMessage> {
        @Override
        protected void updateItem(ChatMessage item, boolean empty) {
            super.updateItem(item, empty);

            if (empty || item == null) {
                setGraphic(null);
                setText(null);
                return;
            }

            boolean out = item.outgoingProperty().get();

            Label title = new Label(out ? "You" : item.fromUserProperty().get());
            title.getStyleClass().add("bubble-title");

            Node bodyNode;
            Label meta;

            if (item.typeProperty().get() == ChatMessage.Type.TEXT) {
                Label body = new Label(item.textProperty().get());
                body.setWrapText(true);
                bodyNode = body;

                meta = new Label(item.timestampProperty().get());
                meta.getStyleClass().add("bubble-meta");

            } else {
                String fileName = item.fileNameProperty().get();

                // If it's an image and downloaded -> show thumbnail
                boolean isImage = FileTypeUtil.kindOf(fileName) == FileTypeUtil.Kind.IMAGE;
                Path local = item.localPathProperty().get();

                if (isImage && local != null && local.toFile().exists()) {
                    ImageView iv = new ImageView(new Image(local.toUri().toString(), 260, 260, true, true, true));
                    iv.setPreserveRatio(true);
                    iv.setSmooth(true);
                    iv.setFitWidth(260);
                    iv.setFitHeight(260);

                    Label caption = new Label(fileName);
                    caption.setWrapText(true);

                    VBox imgBox = new VBox(8, iv, caption);
                    bodyNode = imgBox;
                } else {
                    Label body = new Label(fileName);
                    body.setWrapText(true);

                    Node badge = fileBadgeFor(fileName);
                    HBox fileLine = new HBox(8, badge, body);
                    fileLine.setAlignment(Pos.CENTER_LEFT);
                    bodyNode = fileLine;
                }

                String metaText = humanSize(item.fileSizeProperty().get())
                        + "  •  " + item.timestampProperty().get()
                        + (item.downloadableProperty().get() ? "  •  double-click to download" : "");
                meta = new Label(metaText);
                meta.getStyleClass().add("bubble-meta");
            }

            VBox box = new VBox(4, title, bodyNode, meta);

            StackPane bubble = new StackPane(box);
            bubble.getStyleClass().add("bubble");
            bubble.getStyleClass().add(out ? "bubble-out" : "bubble-in");

            HBox row = new HBox(bubble);
            row.setAlignment(out ? Pos.CENTER_RIGHT : Pos.CENTER_LEFT);

            setGraphic(row);
        }

        private static Node fileBadgeFor(String fileName) {
            Label b = new Label(switch (FileTypeUtil.kindOf(fileName)) {
                case IMAGE -> "IMG";
                case ARCHIVE -> "BOOK";
                case OTHER -> "FILE";
            });
            b.getStyleClass().add("file-badge");
            return b;
        }

        private static String humanSize(long bytes) {
            double b = bytes;
            String[] units = {"B", "KB", "MB", "GB"};
            int i = 0;
            while (b >= 1024 && i < units.length - 1) {
                b /= 1024;
                i++;
            }
            return String.format("%.1f %s", b, units[i]);
        }
    }

    // ... existing code ...
}
```

## ماذا ستحصل الآن؟
- إذا الملف صورة:
    - قبل التنزيل: Bubble فيها “IMG + اسم الملف”
    - بعد التنزيل: Bubble تتحول تلقائيًا لعرض **صورة مصغرة** داخل المحادثة
- إذا rar/zip/7z: يظهر Badge “BOOK”
- غير ذلك: “FILE”
